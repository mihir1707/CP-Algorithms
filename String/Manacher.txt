// Bounding Box Idea

class Manacher {
    vector<int> p;
    
public:
    Manacher(string s){
        build(s);
    }

    void build(string s){
        string t;
        for(auto& ch:s){
            t+='#';
            t+=ch;
        }
        run_manacher(t+'#');
    }

    void run_manacher(string s){
        int n=s.size();
        p.assign(n,0);
        int l=0,r=0;
        for(int i=0;i<n;++i){
            if(i<r){
                p[i]=min(r-i,p[l+r-i]);
            }
            while(i-p[i]-1>=0 && i+p[i]+1<n && s[i+p[i]+1]==s[i-p[i]-1]){
                p[i]++;
            }
            if(i+p[i]>r){
                l=i-p[i];
                r=i+p[i];
            }
        }
    }

    int getLongestPalindrome(int center,bool odd){
        int pos=2*center+1+(!odd);
        return p[pos];
    }

    bool checkPalindrome(int l,int r){
        int center=(l+r)/2;
        bool odd=(r-l)%2==0;
        return getLongestPalindrome(center,odd)>=(r-l+1);
    }

};
