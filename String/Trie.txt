struct Node{
    Node* links[10]={nullptr};
    bool flag=false;
    bool containKey(char ch){
        return links[ch-'0']!=nullptr;
    }
    void put(char ch,Node* node){
        links[ch-'0']=node;
    }
    Node* get(char ch){
        return links[ch-'0'];
    }
    void setEnd(){
        flag=true;
    }
    bool isEnd(){
        return flag;
    }
    ~Node(){
        for(int i=0;i<10;++i){
            if(links[i]!=nullptr){
                delete links[i];
                links[i]=nullptr;
            }
        }
    }
};
class Trie{
    Node* root;
public:
    Trie(){
        root=new Node();
    }
    ~Trie(){
        delete root;
    }
    void insert(string s){
        Node* node=root;
        for(auto& ch:s){
            if(!node->containKey(ch)){
                node->put(ch,new Node());
            }
            node=node->get(ch);
        }
        node->setEnd();
    }
    int f(string s){
        Node* node=root;
        int cnt=0;
        for(auto& ch:s){
            if(!node->containKey(ch)) return cnt;
            node=node->get(ch);
            cnt++;
        }
        return cnt;
    }
};
