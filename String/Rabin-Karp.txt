// for string matching (using hashing)

vector<int> RabinKarp(string &pattern, string &text){
    int n=pattern.size();
    int m=text.size();

    const int p=31;
    const int mod=1e9+9;

    vector<long long> power(max(n,m)+1);
    
    power[0]=1;
    for(int i=1;i<power.size();++i){
        power[i]=(power[i-1]*p)%mod;
    }

    vector<long long> hash(m+1,0);
    for(int i=0;i<m;++i){
        hash[i+1]=(hash[i]+(text[i]-'a'+1)*power[i])%mod;
    }

    long long val=0;
    for(int i=0;i<n;++i){
        val=(val+(pattern[i]-'a'+1)*power[i])%mod;
    }

    vector<int> ans;
    for(int i=0;i<m-n+1;++i){
        long long curr=(hash[i+n]-hash[i]+mod)%mod;
        if(curr==(val*power[i])%mod){
            ans.push_back(i);
        }
    }
}
