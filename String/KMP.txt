// Pattern Matching (using LPS)
// LPS -> longest prefix which is also suffix

class KMP{
    string pat;
    vector<int> lps;

public:
    KMP(string p){
        pat=p;
        build();
    }

    void build(){
        int n=pat.size();
        lps.assign(n,0);
        
        for(int i=1;i<n;++i){
            int j=lps[i-1];
            
            while(j>0 && pat[i]!=pat[j]){
                j=pat[j-1];
            }

            if(pat[i]==pat[j]){
                j++;
            }

            lps[i]=j;
        }
    }

    vector<int> search(string text){
        vector<int> pos;
        int n=text.size();
        int m=pat.size();

        for(int i=0,j=0;i<n;++i){
            while(j>0 && text[i]!=pat[j]){
                j=lps[j-1];
            }

            if(text[i]==pat[j]){
                j++;
            }

            if(j==m){
                pos.push_back(i-m+1);
                j=lps[j-1];
            }
        }

        return pos;
    }
};
