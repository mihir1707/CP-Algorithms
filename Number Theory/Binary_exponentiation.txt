// a<=1e9 , b<=1e9 , mod<=1e9
// if a<=1e18 ((a%mod)^b)%mod;
// if mod<=1e18 need to binary Multiplication
// if b>=1e9 need to ETF & Euler's Theorem

int Binary_Expo(int a,int b){
    int ans=1;
    while(b){
        if(b&1) ans=(ans*a)%mod;
        // ans=Binary_Multi(ans,a)
        a=(a*a)%mod;
        // a=Binary_Multi(a,a)
        b>>=1;
    }
    return ans;
}

int Binary_Multi(long long a,long long b){
    int ans=0;
    while(b){
        if(b&1) ans=(ans+a)%mod;
        a=(a+a)%mod;
        b>>=1;
    }
}
